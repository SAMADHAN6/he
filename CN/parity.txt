
#include <iostream>
using namespace std;

int main() {
    string data, received;
    int count = 0, parity;

    cout << "Enter binary data: ";
    cin >> data;

    // Count number of 1s
    for (int i = 0; i < data.length(); i++) {
        if (data[i] == '1')
            count++;
    }

    // Generate even parity bit
    parity = count % 2;

    cout << "Parity Bit: " << parity << endl;
    cout << "Transmitted Data: " << data << parity << endl;

    // Receive data
    cout << "Enter received data: ";
    cin >> received;

    count = 0;

    // Count 1s in received data
    for (int i = 0; i < received.length(); i++) {
        if (received[i] == '1')
            count++;
    }

    // Check parity
    if (count % 2 == 0)
        cout << "No Error Detected";
    else
        cout << "Error Detected";

    return 0;
}
