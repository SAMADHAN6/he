
#include <iostream>
#include <unistd.h>
#include <sys/socket.h>
#include <arpa/inet.h>
using namespace std;

int main() {
    int s;
    char b[100] = {0};

    // Create TCP socket
    s = socket(AF_INET, SOCK_STREAM, 0);

    // Server address
    sockaddr_in a{};
    a.sin_family = AF_INET;
    a.sin_port = htons(8080);
    inet_pton(AF_INET, "127.0.0.1", &a.sin_addr);

    // Connect to server
    connect(s, (struct sockaddr*)&a, sizeof(a));

    // Send message
    send(s, "Hello Server", 13, 0);

    // Receive reply
    recv(s, b, sizeof(b)-1, 0);
    cout << "Server: " << b << endl;

    // Close socket
    close(s);

    return 0;
}
