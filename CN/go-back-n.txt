#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int n, window;

    cout << "How many frames you want to send: ";
    cin >> n;

    cout << "Enter the window size: ";
    cin >> window;

    if (n <= 0 || window <= 0) {
        cout << "Invalid input!";
        return 0;
    }

    int i = 1;

    while (i <= n) {
        int end = min(i + window - 1, n);

        cout << "Sending frames: ";
        for (int j = i; j <= end; j++) {
            cout << j << " ";
        }
        cout << endl;

        int error;
        cout << "Enter the error no (0 if no error): ";
        cin >> error;

        if (error == 0) {
            cout << "All window successfully done" << endl;
            i = end + 1;
        }
        else if (error >= i && error <= end) {
            cout << "Problem at frame: " << error << endl;
            cout << "Resending frames from: " << error
                 << " to " << end << endl;

            i = error;
        }
        else {
            cout << "Invalid frame number! Try again." << endl;
        }
    }

    cout << "\nAll frames transmitted successfully!";

    return 0;
}
