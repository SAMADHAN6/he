#include <iostream>
using namespace std;

int main() {
    int n;
    int cost[10][10], dist[10], visited[10] = {0};

    cout << "Enter number of routers: ";
    cin >> n;

    cout << "Enter cost matrix (999 for no connection):\n";

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cin >> cost[i][j];
        }
    }

    int start;
    cout << "Enter starting router (0 to " << n-1 << "): ";
    cin >> start;

    for (int i = 0; i < n; i++) {
        dist[i] = cost[start][i];
    }

    dist[start] = 0;
    visited[start] = 1;

    // Find shortest paths using Dijkstra
    for (int count = 1; count < n; count++) {
        int min = 999, u = -1;

        for (int i = 0; i < n; i++) {
            if (!visited[i] && dist[i] < min) {
                min = dist[i];
                u = i;
            }
        }

        if (u == -1)
            break;

        visited[u] = 1;

        for (int v = 0; v < n; v++) {
            if (!visited[v] && dist[u] + cost[u][v] < dist[v]) {
                dist[v] = dist[u] + cost[u][v];
            }
        }
    }

    cout << "\nShortest distances from Router " << start << ":\n";

    for (int i = 0; i < n; i++) {
        cout << "Router " << i << " : " << dist[i] << endl;
    }

    return 0;
}
