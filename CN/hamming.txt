#include <iostream>
using namespace std;

int main(){
    
    int d[8],s1,s2,s3;

    cout<<"Enter pirity bits (eg: at pos 3 5 6 7): ";
    cin>>d[3]>>d[5]>>d[6]>>d[7];

    d[1]=d[3]^d[5]^d[7];
    d[2]=d[3]^d[6]^d[7];
    d[4]=d[5]^d[6]^d[7];

    cout<<"Hamming code: ";
    for(int i=1;i<=7;i++){
        cout<<d[i];
    }
    cout<< endl;
    cout<<"Recived Bits: ";
    for(int i=1;i<=7;i++){
        cin>>d[i];
    }

    s1=d[1]^d[3]^d[5]^d[7];
    s2=d[2]^d[3]^d[6]^d[7];
    s3=d[4]^d[5]^d[6]^d[7];

    int error=s1+(s2*2)+(s3*4);

    if (error == 0) {
        cout << "No Error Detected";
    }
    else {
        cout << "Error at position: " << error;

        // Correct error bit
        d[error] = d[error] ^ 1;

        cout << "\nCorrected Hamming Code: ";

        for (int i = 1; i <= 7; i++)
            cout << d[i];
    }



}
