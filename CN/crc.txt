#include <iostream>
using namespace std;

int main(){
    int n,m,bit[10],divisor[10],temp[10];
    cout <<"Enter the number of bits: ";
    cin>>n;
    for(int i=0;i<n;i++){
        cin>>bit[i];
        temp[i]=bit[i];

    }


    cout <<"Enter the number of divisor: ";
    cin>>m;
    for(int i=0;i<m;i++){
        cin>>divisor[i];
        

    }

    for(int i=n;i<n+m-1;i++){
        temp[i]=0;
    }

    for(int i=0;i<n;i++){
        if(temp[i]==1){
            for(int j=0;j<m;j++){
                temp[i+j]=temp[i+j]^divisor[j];
            }
        }
    }
    cout<<"Changed bits: ";

    for(int i=n;i<n+m-1;i++){
        cout<<temp[i];
    }
    cout<<endl;

    cout<<"Fibnal CRC bits: ";

    for(int i=0;i<n;i++){
        cout<<bit[i];
    }
    for(int i=n;i<n+m-1;i++){
        cout<<temp[i];
    }

}
