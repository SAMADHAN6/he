#include <iostream>
using namespace std;

int main() {
    int n, cost[10][10], dist[10][10];

    cout << "Enter number of routers: ";
    cin >> n;

    cout << "Enter cost matrix (999 for no connection):\n";
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++) {
            cin >> cost[i][j];
            dist[i][j] = cost[i][j];
        }

    // Update shortest distances
    for (int k = 0; k < n; k++)
        for (int i = 0; i < n; i++)
            for (int j = 0; j < n; j++)
                if (dist[i][k] != 999 && dist[k][j] != 999 &&
                    dist[i][k] + dist[k][j] < dist[i][j])
                    dist[i][j] = dist[i][k] + dist[k][j];

    cout << "\nDistance Vector Routing Table:\n";
    for (int i = 0; i < n; i++) {
        cout << "Router " << i << ": ";
        for (int j = 0; j < n; j++)
            cout << dist[i][j] << " ";
        cout << endl;
    }

    return 0;
}
